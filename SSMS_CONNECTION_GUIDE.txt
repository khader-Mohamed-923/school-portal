│  🔍 USEFUL SQL QUERIES                                                     │
└────────────────────────────────────────────────────────────────────────────┘

-- View all students
USE SchoolPortalDB;
SELECT * FROM Students;

-- View all grades
SELECT * FROM Grades;

-- Count records
SELECT COUNT(*) AS TotalStudents FROM Students;
SELECT COUNT(*) AS TotalGrades FROM Grades;

-- View students with grades
SELECT 
    s.FirstName + ' ' + s.LastName AS StudentName,
    s.Email,
    g.CourseName,
    g.Score,
    g.GradeDate
FROM Students s
LEFT JOIN Grades g ON s.Id = g.StudentId
ORDER BY s.Id;

-- View recent students
SELECT TOP 10 * 
FROM Students 
ORDER BY EnrollmentDate DESC;

Problem: Can't connect to localhost,1433
Solution: Make sure Docker containers are running
          Run: docker ps

Problem: Login failed for 'sa'
Solution: Password is: YourStrong!Passw0rd (case-sensitive)

Problem: Database 'SchoolPortalDB' doesn't exist
Solution: Containers need to run at least once to create it
          Run: docker-compose up -d
